// This function sets up the CAN bus using Arduino CAN and provides some error handling for CAN bus errors.  TX and RX queues
// are used to receive and transmit data from the CAN controller in the Arduino minima
static void CAN_ctrl_task(void *args)
{
    CANMessages tx_msg;
    CanMsg rx_message; // Might require const

    xSemaphoreTake(CAN_ctrl_task_sem, portMAX_DELAY);

    if (!CAN.begin(CanBitRate::BR_500k))
    {
        Serial.println("CAN.begin(...) failed.");
        for (;;)
        {
        }
    }
    Serial.println("CAN Driver installed");
    // vTaskDelay(pdMS_TO_TICKS(50));
    Serial.println("CAN Driver started");

    // clear out any old messages on bus
    unsigned long currentTime = millis();
    while ((millis() - currentTime) < 100)
    {
        if (CAN.available())
        {
            rx_message = CAN.read();
        }
        // vTaskDelay(1);
    }

    while (1)
    {
        if (CAN.available())
        {
            rx_message = CAN.read();

            Serial.println(rx_message);

            if (uxQueueSpacesAvailable(CAN_RX_queue))
            {
                xQueueSend(CAN_RX_queue, (void *)&rx_message, 10);
            }
            else
            {
                xQueueSend(CAN_RX_queue, (void *)&rx_message, 10);
                Serial.println(" RX Queue Full");
            }
        }

        if (xQueueReceive(CAN_TX_queue, (void *)&tx_msg, 0) == pdTRUE)
        {
            if (int const rc = CAN.write(tx_msg.message); rc < 0)
            {
                Serial.print("CAN.write(...) failed with error code ");
                Serial.println("rc");
            }
        }

        // if (CAN.isError())
        // {
        //     CAN.clearError();
        //     Serial.println("Error cleared");
        // }
        // Serial.println("CAN CTRL TASK");
        vTaskDelay(100 / portTICK_PERIOD_MS);
    }
    xSemaphoreGive(CAN_ctrl_task_sem);
    vTaskDelete(nullptr);
}

CO_SDO_abortCode_t read_SDO(CO_SDOclient_t *SDO_C, uint8_t nodeID, uint16_t index, uint8_t subIndex, uint8_t *buf, size_t bufSize, size_t *readSize)
{
    CO_SDO_return_t SDO_ret;
    const unsigned int retryAttempts = 5;

    SDO_ret = CO_SDOclient_setup(SDO_C, CO_CAN_ID_SDO_CLI + nodeID, CO_CAN_ID_SDO_SRV + nodeID, nodeID);
    if (SDO_ret != CO_SDO_RT_ok_communicationEnd)
    {
        return CO_SDO_AB_GENERAL;
    }

    SDO_ret = CO_SDOclientUploadInitiate(SDO_C, index, subIndex, 1000, false);
    if (SDO_ret != CO_SDO_RT_ok_communicationEnd)
    {
        return CO_SDO_AB_GENERAL;
    }

    do
    {
        uint32_t timeDifference_us = 1000;
        CO_SDO_abortCode_t abortCode = CO_SDO_AB_NONE;

        SDO_ret = CO_SDOclientUpload(SDO_C, timeDifference_us, false, &abortCode, nullptr, nullptr, nullptr);

        if (SDO_ret < 0)
        {
            CO_SDOclientClose(SDO_C);
            Serial.println("CAN SDO Read");
            return abortCode;
        }

        vTaskDelay(pdMS_TO_TICKS(1));
    } while (SDO_ret > 0);

    *readSize = CO_SDOclientUploadBufRead(SDO_C, buf, bufSize);
    CO_SDOclientClose(SDO_C);
    return CO_SDO_AB_NONE;
}

CO_SDO_abortCode_t write_SDO(CO_SDOclient_t *SDO_C, uint8_t nodeID, uint16_t index, uint8_t subIndex, uint8_t *data, size_t dataSize)
{
    CO_SDO_return_t SDO_ret;
    bool_t bufferPartial = false;

    // setup client (this can be skipped, if remote device is the same)

    SDO_ret = CO_SDOclient_setup(SDO_C, CO_CAN_ID_SDO_CLI + nodeID, CO_CAN_ID_SDO_SRV + nodeID, nodeID);

    if (SDO_ret != CO_SDO_RT_ok_communicationEnd)
    {
        return CO_SDO_AB_NO_DATA;
    }

    // initiate download

    SDO_ret = CO_SDOclientDownloadInitiate(SDO_C, index, subIndex, dataSize, 1000, false);
    if (SDO_ret != CO_SDO_RT_ok_communicationEnd)
    {
        return CO_SDO_AB_NO_DATA;
    }

    // fill data

    size_t nWritten = CO_SDOclientDownloadBufWrite(SDO_C, data, dataSize);
    if (nWritten < dataSize)
    {
        bufferPartial = true;
    }

    // download data
    do
    {
        uint32_t timeDifference_us = 1000;
        CO_SDO_abortCode_t abortCode = CO_SDO_AB_NONE;

        SDO_ret = CO_SDOclientDownload(SDO_C, timeDifference_us, false, bufferPartial, &abortCode, nullptr, nullptr);

        if (SDO_ret < 0)
        {
            Serial.println("CANSDO Read");
            return abortCode;
        }

        vTaskDelay(pdMS_TO_TICKS(1));
    } while (SDO_ret > 0);

    CO_SDOclientClose(SDO_C);
    return CO_SDO_AB_NONE;
}

// CO_main is also provided by CANopenNode in example documentation for basic implementation of CANopenNode.  This function
// has been modified to start the CO_tmr and CO_interrupt tasks that handle processing of messages and receiving of messages

static void CO_main(void *arg)
{
    CO_ReturnError_t err;
    CO_NMT_reset_cmd_t reset = CO_RESET_NOT;
    uint32_t heapMemoryUsed;
    void *CANptr = nullptr;
    uint8_t pendingNodeId = CO_NODE_ID;
    uint8_t activeNodeId = CO_NODE_ID;
    uint16_t pendingBitRate = 500;

    // Allocate memory

    CO_config_t *config_ptr = nullptr;
    CO = CO_new(config_ptr, &heapMemoryUsed);
    if (CO == nullptr)
    {
        Serial.println("Can't allocate memory");
        return;
    }
    else
    {
        Serial.println("Allocated memory for CANopen objects");
    }

    while (reset != CO_RESET_APP)
    {
        /* CANopen communication reset - initialize CANopen objects *******************/
        Serial.println("CANopenNode - Reset Communication...");

        /* Wait rt_thread. */
        CO->CANmodule->CANnormal = false;

        /* Enter CAN configuration. */
        CO_CANsetConfigurationMode((void *)&CANptr);
        CO_CANmodule_disable(CO->CANmodule);

        /* initialize CANopen */
        err = CO_CANinit(CO, CANptr, pendingBitRate);

        if (err != CO_ERROR_NO)
        {
            Serial.println("Error: CAN initialization failed");
            return;
        }

        CO_LSS_address_t lssAddress = {.identity = {
                                           .vendorID = OD_PERSIST_COMM.x1018_identity.vendor_ID,
                                           .productCode = OD_PERSIST_COMM.x1018_identity.productCode,
                                           .revisionNumber = OD_PERSIST_COMM.x1018_identity.revisionNumber,
                                           .serialNumber = OD_PERSIST_COMM.x1018_identity.serialNumber}};

        err = CO_LSSinit(CO, &lssAddress, &pendingNodeId, &pendingBitRate);
        if (err != CO_ERROR_NO)
        {
            Serial.println("Error: LSS slave initialization failed");
            return;
        }

        activeNodeId = pendingNodeId;
        uint32_t errInfo = 0;

        err = CO_CANopenInit(CO,                                         /* CANopen object */
                             nullptr,                                    /* alternate NMT */
                             nullptr,                                    /* alternate em */
                             OD,                                         /* Object dictionary */
                             OD_STATUS_BITS,                             /* Optional OD_statusBits */
                             static_cast<CO_NMT_control_t>(NMT_CONTROL), /* CO_NMT_control_t */
                             FIRST_HB_TIME,                              /* firstHBTime_ms */
                             SDO_SRV_TIMEOUT_TIME,                       /* SDOserverTimeoutTime_ms */
                             SDO_CLI_TIMEOUT_TIME,                       /* SDOclientTimeoutTime_ms */
                             SDO_CLI_BLOCK,                              /* SDOclientBlockTransfer */
                             activeNodeId,
                             &errInfo);
        if (err != CO_ERROR_NO && err != CO_ERROR_NODE_ID_UNCONFIGURED_LSS)
        {
            if (err == CO_ERROR_OD_PARAMETERS)
            {
                Serial.println("Error: Object Dictionary entry");
            }
            else
            {
                Serial.println("Error: CANopen initialization failed");
            }
            return;
        }

        err = CO_CANopenInitPDO(CO, CO->em, OD, activeNodeId, &errInfo);
        if (err != CO_ERROR_NO)
        {
            if (err = CO_ERROR_OD_PARAMETERS)
            {
                Serial.println("Error: Object Dictionary entry");
            }
            else
            {
                Serial.println("Error: PDO initialization failed");
            }
            return;
        }

        /* Configure Timer interrupt function for execution every 1 millisecond */
        auto const co_tmr_create = xTaskCreate(CO_tmrTask, static_cast<const char *>("CO_TMR"), 256, nullptr, CANOPEN_TMR_TASK_PRIO, &co_tmr_task);
        if (co_tmr_create != pdPASS)
            Serial.println("Failed to create 'co_tmr' thread");
        else
            Serial.println("Created co_tmr thread");

        auto const co_int_create = xTaskCreate(CO_interruptTask, static_cast<const char *>("CO_INT"), 256, nullptr, CANOPEN_INTERRUPT_TASK_PRIO, &co_interrupt_task);
        if (co_int_create != pdPASS)
            Serial.println("Failed to create 'co_int' thread");
        else
            Serial.println("Created co_int thread");

        if (!CO->nodeIdUnconfigured)
        {
        }
        else
        {
            Serial.println("CANopenNode - Node-ID not initialized");
        }

        CO_CANsetNormalMode(CO->CANmodule);

        reset = CO_RESET_NOT;

        Serial.println("CANOpenNode - Running...");
        fflush(stdout);

        while (reset == CO_RESET_NOT)
        {
            uint32_t timeDifference_us = 1000;

            reset = CO_process(CO, false, timeDifference_us, nullptr);

            // OD_PERSIST_COMM.x6000_counter++;

            // Serial.println(OD_PERSIST_COMM.x6000_counter);

            if (OD_PERSIST_COMM.x6001_counterFromNode != 0)

            {

                Serial.println("Received RPDO");

                Serial.println(OD_PERSIST_COMM.x6001_counterFromNode);
            }

            /* Nonblocking application code may go here. */

            /* Process automatic storage */

            /* optional sleep for short time */

            // Serial.println("Inside CO PROCESS");

            vTaskDelay(1 / portTICK_PERIOD_MS);
        }
    }

    CO_CANsetConfigurationMode((void *)&CANptr);
    CO_delete(CO);

    Serial.println("CANopenNode finished");
    vTaskDelete(nullptr);
}

static void CO_tmrTask(void *arg)
{
    static unsigned long prevTime = 0;
    while (1)
    {
        if (!CO->nodeIdUnconfigured && CO->CANmodule->CANnormal)
        {
            bool_t syncWas = false;
            /* get time difference since last function call */
            uint32_t timeDifference_us = 1000;

#if (CO_CONFIG_SYNC) & CO_CONFIG_SYNC_ENABLE
            syncWas = CO_process_SYNC(CO, timeDifference_us, nullptr);
#endif
#if (CO_CONFIG_PDO) & CO_CONFIG_RPDO_ENABLE
            CO_process_RPDO(CO, syncWas, timeDifference_us, nullptr);
            // Serial.println("RPDO Process");
#endif
#if (CO_CONFIG_PDO) & CO_CONFIG_TPDO_ENABLE
            CO_process_TPDO(CO, syncWas, timeDifference_us, nullptr);
            // Serial.println("PDO Process Running");
#endif
            /* Further I/O or nonblocking application code may go here. */
        }
        vTaskDelay(10 / portTICK_PERIOD_MS);
    }
    vTaskDelete(nullptr);
}

static void CO_interruptTask(void *arg)
{
    while (1)
    {
        CO_CANinterrupt(CO->CANmodule);
        // Serial.println("Interrupt task");
        vTaskDelay(10 / portTICK_PERIOD_MS);
    }

    vTaskDelete(nullptr);
}

            pastTime = currentTime;
            counter++;
